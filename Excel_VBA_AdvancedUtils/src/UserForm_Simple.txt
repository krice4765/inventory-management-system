Option Explicit

Private Sub UserForm_Initialize()
    Me.Caption = "AdvancedUtils 簡易設定"
    
    optDuplicate.Caption = "重複ファイル検出"
    optDisk.Caption = "ディスク使用量分析"
    optOrganize.Caption = "ファイル整理・分類"
    
    optDuplicate.Value = True
    
    btnExecute.Caption = "実行"
    btnCancel.Caption = "キャンセル"
    
    txtFolderPath.Text = CreateObject("WScript.Shell").SpecialFolders("Desktop")
End Sub

Private Sub btnBrowse_Click()
    Dim folderPath As String
    folderPath = BrowseForFolder("処理対象フォルダを選択")
    If folderPath <> "" Then
        txtFolderPath.Text = folderPath
    End If
End Sub

Private Sub btnExecute_Click()
    Dim selectedPath As String
    selectedPath = txtFolderPath.Text
    
    If selectedPath = "" Then
        MsgBox "フォルダを選択してください", vbExclamation
        Exit Sub
    End If
    
    Me.Hide
    
    If optDuplicate.Value Then
        Module_DuplicateDetector.DetectDuplicatesMain selectedPath
    ElseIf optDisk.Value Then
        Module_DiskAnalyzer.AnalyzeDiskMain selectedPath
    ElseIf optOrganize.Value Then
        Module_FileOrganizer.OrganizeFilesMain selectedPath
    End If
    
    Unload Me
End Sub

Private Sub btnCancel_Click()
    Unload Me
End Sub
